const { chromium } = require('/workspace/GameAI-Hub-b3-prod/node_modules/playwright');
const fs = require('node:fs');
const dir='/workspace/GameAI-Hub-b3-prod/docs/screenshots/b3-home/independent-visual';
const base=process.env.B3_REVIEW_URL||'http://127.0.0.1:3179';
(async()=>{
 const b=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});const c=await b.newContext();await c.route('**/*',r=>new URL(r.request().url()).hostname==='127.0.0.1'?r.continue():r.abort());await c.addInitScript(()=>localStorage.setItem('gameai:analytics-excluded','1'));const p=await c.newPage();const samples=[];
 for(const [width,section] of [[1440,'home'],[375,'home'],[320,'home'],[375,'next'],[375,'create'],[375,'create-expanded'],[375,'categories'],[375,'menu'],[375,'preview'],[375,'continuation'],[375,'featured'],[375,'examples'],[375,'trust'],[375,'footer']]){
  await p.setViewportSize({width,height:width===1440?900:667});await p.goto(`${base}/`);await p.waitForFunction(()=>document.querySelector('.b3-showcase')?.dataset.mode==='deck'&&document.querySelector('.b3-game-stage')?.dataset.motion==='idle');await p.evaluate(()=>document.fonts.ready);
  const sectionSelectors={next:'.b3-next-chapter',create:'.home-execution-hero','create-expanded':'.home-execution-hero',categories:'.home-category-entry',preview:'.home-example',continuation:'.home-continuation',featured:'.home-featured',examples:'.example-projects',trust:'.home-trust-note',footer:'.site-footer'};if(section==='create-expanded')await p.locator('.home-execution-hero details summary').click();if(sectionSelectors[section])await p.locator(sectionSelectors[section]).evaluate(e=>window.scrollTo({top:e.getBoundingClientRect().top+scrollY-document.querySelector('.site-header').getBoundingClientRect().height-16,behavior:'instant'}));if(section==='menu')await p.getByRole('button',{name:'メニューを開く'}).click();await p.waitForTimeout(150);await p.screenshot({path:`${dir}/contrast-original-${section}-${width}.png`});
  const nodes=await p.evaluate(()=>{
   const selectors=['.site-header .brand-copy','.site-header .header-cta','.site-header .main-nav','.site-header .mobile-menu','.b3-intro','.b3-gallery-bar','.b3-game-stage [data-distance="0"] .b3-art-top','.b3-game-stage [data-distance="0"] .b3-art-credit','.b3-game-selection','.b3-deck-help','.b3-next-chapter','.home-execution-hero','.home-category-entry','.home-example','.home-continuation','.home-featured','.example-projects','.home-trust-note','.site-footer'];const result=[];const headerBottom=document.querySelector('.site-header .header-inner').getBoundingClientRect().bottom;
   for(const selector of selectors)for(const element of document.querySelectorAll(selector)){const walker=document.createTreeWalker(element,NodeFilter.SHOW_TEXT);while(walker.nextNode()){const node=walker.currentNode;if(!node.textContent.trim())continue;const parent=node.parentElement;const style=getComputedStyle(parent);if(parent.closest('.sr-only,[hidden],[inert]')||style.visibility==='hidden'||style.opacity==='0')continue;if(parent.closest('details:not([open])')&&!parent.closest('summary'))continue;const item=parent.closest('.article-cluster-list>li');if(item?.dataset.distance&&item.dataset.distance!=='0')continue;const range=document.createRange();range.selectNodeContents(node);const rects=[...range.getClientRects()].filter(r=>r.width>0&&r.height>0&&r.left>=0&&r.right<=innerWidth&&r.top>=(parent.closest('.site-header')?0:headerBottom)&&r.bottom<=innerHeight).map(r=>({left:r.left,top:r.top,right:r.right,bottom:r.bottom}));if(!rects.length)continue;result.push({selector,text:node.textContent.trim(),color:style.color,fontSize:parseFloat(style.fontSize),fontWeight:style.fontWeight,rects});}}
   return result;
  });
  await p.addStyleTag({content:'.site-header *,.home-b3 *,.site-footer * {color:transparent!important;text-shadow:none!important;transition:none!important}'});await p.waitForTimeout(50);const name=`contrast-bg-${section}-${width}`;await p.screenshot({path:`${dir}/${name}.png`});samples.push({name,width,section,nodes});console.log(name,nodes.length);
 }
 fs.writeFileSync(`${dir}/contrast-samples.json`,JSON.stringify({method:'Rendered background pixel sampling; disposable text-transparent browser styles; no source edits',server:base,samples},null,2));await b.close();
})().catch(e=>{console.error(e);process.exit(1)});
